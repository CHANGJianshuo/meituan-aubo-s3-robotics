cmake_minimum_required(VERSION 3.8)
project(mtc_ros2_action_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(action_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(mtc_interfaces REQUIRED)
find_package(mtc_core_cpp REQUIRED)
find_package(mtc_task_cpp REQUIRED)
find_package(mtc_manipulation_cpp REQUIRED)
find_package(mtc_motion_execution_cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(${PROJECT_NAME} SHARED src/mock_ports.cpp src/pick_place_server.cpp
  src/task_server.cpp src/motion_server.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} PUBLIC rclcpp rclcpp_action action_msgs std_msgs mtc_interfaces)
target_link_libraries(${PROJECT_NAME} PUBLIC
  mtc_task_cpp::mtc_task_cpp mtc_manipulation_cpp::mtc_manipulation_cpp
  mtc_motion_execution_cpp::mtc_motion_execution_cpp Threads::Threads)
add_executable(mock_pick_place_server src/pick_place_main.cpp)
target_link_libraries(mock_pick_place_server ${PROJECT_NAME})
add_executable(mock_task_server src/task_main.cpp)
target_link_libraries(mock_task_server ${PROJECT_NAME})
add_executable(mock_motion_server src/motion_main.cpp)
target_link_libraries(mock_motion_server ${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} mock_pick_place_server mock_task_server mock_motion_server
  EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_pick_place_action test/test_pick_place_action.cpp TIMEOUT 90)
  target_link_libraries(test_pick_place_action ${PROJECT_NAME})
  ament_add_gtest(test_task_action test/test_task_action.cpp TIMEOUT 90)
  target_link_libraries(test_task_action ${PROJECT_NAME})
  ament_add_gtest(test_motion_action test/test_motion_action.cpp TIMEOUT 90)
  target_link_libraries(test_motion_action ${PROJECT_NAME})
  ament_add_gtest(test_action_contracts test/test_action_contracts.cpp)
  target_link_libraries(test_action_contracts ${PROJECT_NAME})
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(Threads rclcpp rclcpp_action action_msgs std_msgs mtc_interfaces mtc_task_cpp
  mtc_manipulation_cpp mtc_motion_execution_cpp)
ament_package()
