cmake_minimum_required(VERSION 3.16)
project(mtc_simulation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(mtc_moveit_planner REQUIRED)
find_package(mtc_manipulation_cpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(moveit_core REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(mtc_v1_observer SHARED src/observer.cpp)
target_link_libraries(mtc_v1_observer gz-sim8::gz-sim8 gz-plugin2::register nlohmann_json::nlohmann_json)
ament_target_dependencies(mtc_v1_observer rclcpp tf2_msgs std_msgs)
add_library(mtc_sim_planning_only SHARED src/planning_only.cpp)
ament_target_dependencies(mtc_sim_planning_only pluginlib moveit_core)
pluginlib_export_plugin_description_file(moveit_core planning_only.xml)
add_executable(v1_task src/v1_task.cpp)
ament_target_dependencies(v1_task rclcpp rclcpp_action sensor_msgs tf2_msgs std_msgs control_msgs
  controller_manager_msgs mtc_moveit_planner mtc_manipulation_cpp)
target_link_libraries(v1_task nlohmann_json::nlohmann_json)
install(TARGETS mtc_v1_observer mtc_sim_planning_only LIBRARY DESTINATION lib)
install(TARGETS v1_task DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY assets config worlds launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/run_v1.py scripts/prepare_v1.py scripts/analyze_v1.py scripts/record_camera.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_acceptance test/test_acceptance.py)
endif()
ament_package()
